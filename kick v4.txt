local function getPlayerList()
	local t = {}
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= LocalPlayer then
			table.insert(t, p.DisplayName .. " | " .. p.Name)
		end
	end
	table.sort(t)
	return t
end

local function playerFromString(s)
	if not s or s == "" then return nil end
	local name = s:match("%|%s*(.+)$")
	if not name then return nil end
	return Players:FindFirstChild(name)
end

local function zeroPart(p)
	if p then
		p.Velocity = Vector3.zero
		p.AssemblyLinearVelocity = Vector3.zero
		p.RotVelocity = Vector3.zero
		p.AssemblyAngularVelocity = Vector3.zero
	end
end

local function applyGyro(char)
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp then return end
	local gyro = hrp:FindFirstChild("KickHoldGyro")
	if not gyro then
		gyro = Instance.new("BodyGyro")
		gyro.Name = "KickHoldGyro"
		gyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
		gyro.P = 50000
		gyro.D = 100
		gyro.Parent = hrp
	end
	gyro.CFrame = CFrame.new(hrp.Position, hrp.Position + Vector3.new(1, 0, 0))
end

local function toggleParts(char, on)
	if not char then return end
	for _, name in ipairs({"Head","Torso","UpperTorso","LowerTorso","Left Arm","Right Arm","Left Leg","Right Leg"}) do
		local p = char:FindFirstChild(name)
		if p then
			if on then
				zeroPart(p)
				p.CanCollide = false
				p.CanTouch = false
			else
				p.CanCollide = true
				p.CanTouch = true
			end
		end
	end
end

local function applyBody(parent, pos, look)
	local bp = parent:FindFirstChild("KickHoldBP")
	if not bp then
		bp = Instance.new("BodyPosition")
		bp.Name = "KickHoldBP"
		bp.Parent = parent
	end
	bp.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
	bp.P = 50000
	bp.D = 100
	bp.Position = pos

	local bg = parent:FindFirstChild("KickHoldBG")
	if not bg then
		bg = Instance.new("BodyGyro")
		bg.Name = "KickHoldBG"
		bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
		bg.P = 50000
		bg.D = 100
		bg.Parent = parent
	end
	bg.CFrame = CFrame.new(parent.Position, parent.Position + look)
end

local function stopKick()
	kickEnabled = false
	for _, t in ipairs(kickThreads) do
		if t then pcall(task.cancel, t) end
	end
	kickThreads = {}

	for _, player in ipairs(Players:GetPlayers()) do
		local char = player.Character
		if char then
			for _, d in ipairs(char:GetDescendants()) do
				if d.Name:find("KickHold") or d.Name:find("PartOwner") then
					pcall(function() d:Destroy() end)
				end
			end
			for _, partName in ipairs({"Head","Torso","UpperTorso","LowerTorso","HumanoidRootPart","Left Arm","Right Arm","Left Leg","Right Leg"}) do
				local p = char:FindFirstChild(partName)
				if p then
					p.CanCollide = true
					p.CanTouch = true
					p.Velocity = Vector3.zero
					p.AssemblyLinearVelocity = Vector3.zero
				end
			end
		end
	end

	if cFrame then
		local char = LocalPlayer.Character
		if char then
			local hrp = char:FindFirstChild("HumanoidRootPart")
			if hrp then
				hrp.CFrame = cFrame
				hrp.Velocity = Vector3.zero
			end
		end
		cFrame = nil
	end

	workspace.FallenPartsDestroyHeight = fallenPartsDestroyHeight
end

local function kickLoop(target)
	local locked = false
	local saved = nil

	local t1 = task.spawn(function()
		while kickEnabled and target and target.Parent do
			local myChar = LocalPlayer.Character
			local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
			if not myRoot then task.wait() continue end

			local tChar = target.Character
			local tHum = tChar and tChar:FindFirstChild("Humanoid")
			local tRoot = tChar and tChar:FindFirstChild("HumanoidRootPart")
			if not tChar or not tHum or tHum.Health <= 0 or not tRoot then
				if saved then
					myRoot.CFrame = saved
					zeroPart(myRoot)
					saved = nil
					cFrame = nil
				end
				locked = false
				task.wait()
				continue
			end

			toggleParts(tChar, true)
			applyGyro(tChar)

			local owned = false
			for _, d in ipairs(tChar:GetDescendants()) do
				if d.Name == "PartOwner" and d:IsA("StringValue") and d.Value == LocalPlayer.Name then
					owned = true
					break
				end
			end

			if owned then
				if locked then
					locked = false
					if saved then
						myRoot.CFrame = saved
						zeroPart(myRoot)
						saved = nil
						cFrame = nil
					end
				end
			elseif (tRoot.Position - (locked and saved and saved.Position or myRoot.Position)).Magnitude > 30 then
				if not locked then
					locked = true
					saved = myRoot.CFrame
					cFrame = myRoot.CFrame
				end
				myRoot.CFrame = CFrame.new(tRoot.Position) * CFrame.new(0, -15, 0)
				zeroPart(myRoot)
			elseif locked then
				locked = false
				if saved then
					myRoot.CFrame = saved
					zeroPart(myRoot)
					saved = nil
					cFrame = nil
				end
			end

			local basePos = (locked and saved and saved.Position or myRoot.Position) + Vector3.new(offsetX, offsetY, offsetZ)
			applyBody(tRoot, basePos, myRoot.CFrame.LookVector)
			task.wait()
		end
	end)
	table.insert(kickThreads, t1)

	local t2 = task.spawn(function()
		while kickEnabled and target and target.Parent do
			local tRoot = target.Character and target.Character:FindFirstChild("HumanoidRootPart")
			if tRoot then
				pcall(function() SNO:FireServer(tRoot, tRoot.CFrame) end)
			end
			task.wait()
		end
	end)
	table.insert(kickThreads, t2)

	local t3 = task.spawn(function()
		while kickEnabled and target and target.Parent do
			local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
			local tRoot = target.Character and target.Character:FindFirstChild("HumanoidRootPart")
			if myRoot and tRoot then
				local lockPos = (myRoot.Position + Vector3.new(offsetX, offsetY, offsetZ))
				if (tRoot.Position - lockPos).Magnitude <= 1.5 then
					if (tRoot.Position - myRoot.Position).Magnitude < 30 then
						pcall(function() DestroyLine:FireServer(tRoot) end)
					end
				end
			end
			if destroyLineWait > 0 then task.wait(destroyLineWait) else task.wait() end
		end
	end)
	table.insert(kickThreads, t3)
end

local function startKick(target)
	if not target or not target.Parent then return end
	pcall(function()
		stopKick()
		kickEnabled = true
		kickLoop(target)
	end)
end

local pclespEnabled = false
local pclespAdornments = {}
local lastScan = 0
local scanInterval = 1
local smoothFactor = 0.15

local function CreateESP(part)
	if not part or not part.Parent then return end
	local box = Instance.new("BoxHandleAdornment")
	box.Size = part.Size
	box.AlwaysOnTop = true
	box.ZIndex = 5
	box.Transparency = 0.2
	box.Color3 = Color3.fromRGB(138, 43, 226)
	box.Adornee = part
	box.Parent = part

	local outline = Instance.new("SelectionBox")
	outline.LineThickness = 0.02
	outline.Color3 = Color3.new(0, 0, 0)
	outline.Transparency = 0.3
	outline.Adornee = part
	outline.Parent = part

	pclespAdornments[part] = { box = box, outline = outline, currentSize = part.Size }
end

local function UpdateESP(part)
	local data = pclespAdornments[part]
	if not data then return end
	local box = data.box
	if not box or not box.Parent then return end
	local newSize = data.currentSize:Lerp(part.Size, smoothFactor)
	data.currentSize = newSize
	box.Size = newSize
end

local function ScanForESPParts()
	for _, obj in pairs(workspace:GetDescendants()) do
		if obj:IsA("BasePart") and (obj.Name == "partesp" or obj.Name == "playercharacterlocationdetector" or obj.Name == "PlayerCharacterLocationDetector") then
			if not pclespAdornments[obj] then
				CreateESP(obj)
			end
		end
	end
end

local function CleanupESP()
	local toRemove = {}
	for part, data in pairs(pclespAdornments) do
		if not part.Parent then
			table.insert(toRemove, part)
		end
	end
	for _, part in ipairs(toRemove) do
		local data = pclespAdornments[part]
		if data then
			if data.box then data.box:Destroy() end
			if data.outline then data.outline:Destroy() end
		end
		pclespAdornments[part] = nil
	end
end

PCLDBox:AddToggle("PCLD", {
	Text = "PCLD ESP",
	Default = false,
	Callback = function(v)
		pclespEnabled = v
		if v then
			lastScan = 0
		else
			for part, data in pairs(pclespAdornments) do
				if data.box then data.box:Destroy() end
				if data.outline then data.outline:Destroy() end
			end
			pclespAdornments = {}
		end
	end,
})

RunService.Heartbeat:Connect(function()
	if not pclespEnabled then return end
	local now = tick()
	if now - lastScan >= scanInterval then
		lastScan = now
		ScanForESPParts()
		CleanupESP()
	end
	for part, _ in pairs(pclespAdornments) do
		UpdateESP(part)
	end
end)

local autoResetEnabled = false
local autoLeaveEnabled = false
local autoLeaveSensitivity = 3
local warnTimestamps = {}
local gameCorrectionConn = nil

local function getGameCorrectionsEvent()
	local gce = RS:FindFirstChild("GameCorrectionEvents")
	if not gce then
		gce = RS:WaitForChild("GameCorrectionEvents", 5)
	end
	return gce and gce:FindFirstChild("GameCorrectionsNotify")
end

local function startGameCorrectionListener()
	if gameCorrectionConn then
		gameCorrectionConn:Disconnect()
		gameCorrectionConn = nil
	end

	local notifyEvent = getGameCorrectionsEvent()
	if not notifyEvent then return end

	gameCorrectionConn = notifyEvent.OnClientEvent:Connect(function(reason)
		if reason ~= "Flying" then return end

		if autoResetEnabled then
			local char = LocalPlayer.Character
			local hum = char and char:FindFirstChild("Humanoid")
			if hum then
				pcall(function()
					char:BreakJoints()
				end)
				pcall(function()
					hum.Health = 0
				end)
			end
			Library:Notify({ Title = "Auto Reset", Description = "Flying detected, reset.", Time = 3 })
		end

		if autoLeaveEnabled then
			local now = tick()
			table.insert(warnTimestamps, now)
			for i = #warnTimestamps, 1, -1 do
				if now - warnTimestamps[i] > 1.5 then
					table.remove(warnTimestamps, i)
				end
			end
			if #warnTimestamps >= autoLeaveSensitivity then
				LocalPlayer:Kick("Auto Leave: " .. #warnTimestamps .. " warns in short time.")
			end
		end
	end)
end
